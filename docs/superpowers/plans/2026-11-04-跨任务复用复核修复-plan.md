## 每批汇报约定

每批完成后只报三样，便于复核判断：

1. **复现时实际错在哪个格**（坐标 + 当时的值）；
2. **修完后该格是什么值**；
3. **该侧三遍各调用了几次**。

停用与裁决另加三项：批次号、版本状态、受影响清单。

## 测试 helper 约定（各任务共用）

- `_跑真实任务(目录, 名, **变化) -> dict`：**固定模型答复，但提取、写回、回读走真实 Word 桥接**。做法照 `测试/test_真实桥接集成.py::test_三组标题与数值_重新打开核对`：只 `monkeypatch.setattr("核心.识别.调用识别", _固定模型(**变化))`，`核心/提取.py`、`核心/写回.py`、`核心/回读.py` 保持真实，然后调用 `运行任务`。

```python
def _期间核对(日期组, 视图, 上下文, 区域):
    if not 日期组:
        return ""
    本次, 记录 = [], []
    for 项 in 日期组:
        行, 列 = 视图.顶 + 项["相对"][0], 视图.左 + 项["相对"][1]
        文 = 视图.文本(行, 列)
        值 = 解析日期(文) if 文 else None
        if 值 is None:
            return f"日期位置已不是日期：{get_column_letter(列)}{行}"
        旧 = 解析日期(项.get("值") or "")
        if 旧 is None:
            return "记录里的日期无法解析"
        # 角色词必须仍绑在同一列、同一位置、同一原文
        if 项.get("角色词"):
            表头 = 视图.文本(视图.顶 + 项["表头相对"][0], 视图.左 + 项["表头相对"][1])
            if (表头 or "").strip() != 项.get("表头原文"):
                return (f"期间角色文字已变：{get_column_letter(列)} "
                        f"原“{项.get('表头原文')}”现“{表头 or ''}”")
        if (值.month, 值.day) != (旧.month, 旧.day):
            return f"期间月日已变：{get_column_letter(列)}{行} 原 {项['值']} 现 {文}"
        本次.append(值); 记录.append(旧)
    if len({a.year - b.year for a, b in zip(本次, 记录)}) != 1:
        return "整组日期没有按同一年度差平移"
    # 单日期：有角色词按角色核对；无角色词则必须有独立期间依据，且日期要落在该期间的正确端点
    if len(本次) == 1 and not 日期组[0].get("角色词"):
        期间 = _上下文期间(上下文, 区域)          # {"起": date, "止": date} 或 None
        if 期间 is None:
            return "只有一个日期、没有角色词，也没有独立期间依据"
        角色 = 日期组[0].get("期间角色") or ""
        if "期末" in 角色:
            应 = 期间["止"]
        elif "期初" in 角色:
            应 = 期间["起"]
        else:
            return "只有一个日期且角色不明，无法核对"
        if 本次[0] != 应:
            return f"本次日期 {本次[0]} 与上下文期间的{角色} {应} 不符"
    return ""
```

```python
    if not (记录们 or []):

    本次 = _上下文单位(上下文, 区域)          # 表前/章节说明的解析结果，没有则 None
```
